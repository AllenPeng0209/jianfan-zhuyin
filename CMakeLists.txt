cmake_minimum_required(VERSION 3.20)
project(JianfanZhuyinCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
option(JIANFAN_TESTS "Build offline engine tests" ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    add_compile_options(/utf-8 /EHsc /DNOMINMAX /DWIN32_LEAN_AND_MEAN)
endif()
if(JIANFAN_TESTS)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(GOOGLETEST_VERSION 1.11.0)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    add_subdirectory(Vendor/Dependencies/SwiftyOpenCC/OpenCC/deps/gtest-1.11.0 EXCLUDE_FROM_ALL)
    if(NOT TARGET GTest::gtest_main)
        add_library(GTest::gtest_main ALIAS gtest_main)
    endif()
    enable_testing()
endif()
set(ENABLE_TEST ${JIANFAN_TESTS})
add_subdirectory(Vendor/McBopomofo/Source/Engine)
include(cmake/OpenCC.cmake)
add_library(JianfanCore core/Session.cpp core/OfflineStore.cpp)
target_include_directories(JianfanCore PUBLIC core Vendor/McBopomofo/Source/Engine)
target_link_libraries(JianfanCore PUBLIC McBopomofoLMLib gramambular2_lib MandarinLib PRIVATE JianfanOpenCC)
if(JIANFAN_TESTS)
add_executable(JianfanCoreTest tests/core/SessionTest.cpp tests/core/OfflineStoreTest.cpp)
target_link_libraries(JianfanCoreTest PRIVATE JianfanCore GTest::gtest_main)
target_compile_definitions(JianfanCoreTest PRIVATE JIANFAN_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
include(GoogleTest)
gtest_discover_tests(JianfanCoreTest)
endif()
if(WIN32)
    add_subdirectory(platforms/windows)
endif()
